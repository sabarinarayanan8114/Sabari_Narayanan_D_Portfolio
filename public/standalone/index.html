/**
 * Google Apps Script Web App Backend
 * Receives Contact Form Submissions and Appends to Google Sheet Database
 */
function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);

  try {
    var doc = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = doc.getSheetByName('Submissions') || doc.getActiveSheet();
    
    // Auto-create header row if sheet is empty
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(['Timestamp', 'Name', 'Email', 'Subject', 'Message', 'Status']);
      var headerRange = sheet.getRange(1, 1, 1, 6);
      headerRange.setFontWeight('bold');
      headerRange.setBackground('#0f172a');
      headerRange.setFontColor('#38bdf8');
    }

    var data = e.parameter || {};
    // Also parse JSON payload if sent as application/json
    if (e.postData && e.postData.contents) {
      try {
        var jsonData = JSON.parse(e.postData.contents);
        for (var key in jsonData) {
          data[key] = jsonData[key];
        }
      } catch (err) {}
    }

    var timestamp = new Date();
    var name = data.name || 'Anonymous';
    var email = data.email || 'No email provided';
    var subject = data.subject || 'Portfolio Inquiry';
    var message = data.message || '';

    // Append to Google Sheet Database
    sheet.appendRow([timestamp, name, email, subject, message, 'NEW']);

    // Send email notification to Sabari
    try {
      MailApp.sendEmail({
        to: 'dsabari2408@gmail.com',
        subject: '[Portfolio Lead] ' + subject + ' from ' + name,
        body: 'New Contact Form Submission:\n\n' +
              'Name: ' + name + '\n' +
              'Email: ' + email + '\n' +
              'Subject: ' + subject + '\n\n' +
              'Message:\n' + message + '\n\n' +
              'Logged in Google Sheet: ' + doc.getUrl()
      });
    } catch (mailErr) {
      Logger.log('Mail error: ' + mailErr.toString());
    }

    return ContentService.createTextOutput(JSON.stringify({
      result: 'success',
      row: sheet.getLastRow(),
      message: 'Submission successfully recorded in Google Sheets database!'
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({
      result: 'error',
      error: error.toString()
    })).setMimeType(ContentService.MimeType.JSON);

  } finally {
    lock.releaseLock();
  }
}

function doGet(e) {
  return ContentService.createTextOutput(JSON.stringify({
    status: 'online',
    message: 'Google Apps Script Database Web App is active and ready to receive POST requests.'
  })).setMimeType(ContentService.MimeType.JSON);
}

1-Minute Google Sheets & Apps Script Setup:

  1. Create a Google Sheet: Go to sheets.new and name it Portfolio Contact Database.
  2. Open Apps Script: In Google Sheets, click Extensions > Apps Script.
  3. Paste Code: Delete existing code and paste the Code.gs script from the tab above.
  4. Deploy as Web App:
    • Click Deploy > New deployment.
    • Select type: Web app.
    • Description: Portfolio Contact Backend.
    • Execute as: Me.
    • Who has access: Anyone (crucial so visitors can submit).
    • Click Deploy and copy the Web App URL!
  5. Paste Web App URL: In script.js, replace GOOGLE_APPS_SCRIPT_URL with your new URL. All submissions will now save to your Google Sheet in real-time!
